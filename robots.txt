Allow: /landing.html

# index.html (la app) NO se indexa, pero NO se bloquea aquí a propósito: la
# señal es su <meta name="robots" content="noindex, follow">, y para que Google
# la respete tiene que poder RASTREAR la página. Un Disallow impediría que viera
# el noindex y la URL podría acabar listada igualmente. Mismo criterio para
# privacy.html y terms.html (duplicados legales con noindex por meta).

# Rutas y ficheros que no deben rastrearse. Es una capa de INTENCIÓN: la
# seguridad real la aplica el servidor (static-guard devuelve 404 a estos).

Disallow: /src/
Disallow: /admin
Disallow: /.env
Disallow: /.env.production
Disallow: /package.json
